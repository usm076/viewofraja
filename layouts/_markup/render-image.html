{{- /* Images in Markdown content. Images uploaded through Pages CMS live in static/images and are mounted into assets (see hugo.toml), so they can be resized here: responsive WebP sources, explicit dimensions (no layout shift) and lazy loading. Remote or unprocessable images render as-is. A Markdown title becomes the caption: ![alt](/images/x.jpg "Caption") */ -}} {{- $src := .Destination -}} {{- $alt := .Text | plainify -}} {{- $res := false -}} {{- if not (findRE `^(https?:|data:)` $src) -}} {{- with .Page.Resources.Get $src }}{{ $res = . }}{{ else }}{{ with resources.Get (strings.TrimPrefix "/" $src) }}{{ $res = . }}{{ end }}{{ end -}} {{- end -}} {{- /* Trim EOF */ -}}
